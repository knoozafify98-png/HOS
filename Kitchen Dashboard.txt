                <div className="flex items-center justify-between p-4 bg-slate-50 rounded-2xl border border-slate-200">
                  <div className="flex items-center gap-3">
                    {isAnonymous ? <UserX className="w-5 h-5 text-purple-600" /> : <UserCheck className="w-5 h-5 text-emerald-600" />}
                    <div>
                      <div className="text-xs font-bold text-slate-900">Anonymous Mode</div>
                      <div className="text-[11px] text-slate-500">Hide employee identity on submission log</div>
                    </div>
                  </div>
                  <button
                    type="button"
                    onClick={() => setIsAnonymous(!isAnonymous)}
                    className={`w-12 h-6 rounded-full transition-colors p-0.5 relative ${
                      isAnonymous ? 'bg-purple-600' : 'bg-slate-300'
                    }`}
                  >
                    <div
                      className={`w-5 h-5 bg-white rounded-full transition-transform ${
                        isAnonymous ? 'translate-x-6' : 'translate-x-0'
                      }`}
                    ></div>
                  </button>
                </div>

                <button
                  type="submit"
                  className="w-full py-3.5 bg-emerald-600 hover:bg-emerald-700 text-white font-black rounded-2xl text-xs uppercase tracking-wider flex items-center justify-center gap-2 shadow-md transition-colors"
                >
                  <Send className="w-4 h-4" /> Submit Shift Report
                </button>
              </form>
            </div>
          </div>
        )}

        {}
        {/* MODULE 5: INVENTORY */}
        {activeTab === 'inventory' && (
          <div className="space-y-6">
            {/* Storage Tabs */}
            <div className="flex bg-white p-1.5 rounded-2xl border border-slate-200 shadow-xs max-w-lg mx-auto">
              <button
                onClick={() => setInvTab('dry')}
                className={`flex-1 py-2.5 rounded-xl text-xs font-bold transition-all ${
                  invTab === 'dry'
                    ? 'bg-emerald-600 text-white shadow-md'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                📦 Storage Room (Dry Goods)
              </button>
              <button
                onClick={() => setInvTab('chilled')}
                className={`flex-1 py-2.5 rounded-xl text-xs font-bold transition-all ${
                  invTab === 'chilled'
                    ? 'bg-emerald-600 text-white shadow-md'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                ❄️ Refrigerator (Chilled)
              </button>
              <button
                onClick={() => setInvTab('frozen')}
                className={`flex-1 py-2.5 rounded-xl text-xs font-bold transition-all ${
                  invTab === 'frozen'
                    ? 'bg-emerald-600 text-white shadow-md'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                🧊 Freezer (Frozen)
              </button>
            </div>

            {/* Inventory Items List */}
            <div className="bg-white rounded-3xl border border-slate-200 shadow-sm overflow-hidden">
              <div className="p-4 bg-slate-50 border-b border-slate-200 flex items-center justify-between">
                <span className="font-bold text-xs text-slate-700 uppercase tracking-wider">
                  FIFO Expiry Tracking & Allergen Badges
                </span>
                <span className="text-xs text-slate-500 font-semibold">
                  {initialInventory.filter(i => i.category === invTab).length} Items Tracked
                </span>
              </div>

              <div className="divide-y divide-slate-100">
                {initialInventory
                  .filter(i => i.category === invTab)
                  .map((item) => (
                    <div key={item.id} className="p-4 flex flex-col sm:flex-row sm:items-center justify-between gap-3 hover:bg-slate-50 transition-colors">
                      <div>
                        <div className="font-bold text-slate-900 text-sm">{item.name}</div>
                        <div className="text-xs text-slate-500 font-medium mt-0.5">Quantity on Hand: <strong className="text-slate-800">{item.qty}</strong></div>
                      </div>

                      <div className="flex flex-wrap items-center gap-2">
                        {/* Allergen Purple Badges */}
                        {item.allergens.map(allergen => (
                          <span key={allergen} className="bg-purple-100 text-purple-800 font-bold px-2.5 py-1 rounded-md text-xs border border-purple-200">
                            ⚠️ Contains {allergen}
                          </span>
                        ))}

                        {/* FIFO Expiry Badge */}
                        {getExpiryBadge(item.expiry)}
                      </div>
                    </div>
                  ))}
              </div>
            </div>
          </div>
        )}

        {}
        {/* MODULE 6: HANDOVER */}
        {activeTab === 'handover' && (
          <div className="space-y-6">
            {/* Add New Handover Input */}
            <div className="bg-white p-5 rounded-3xl border border-slate-200 shadow-sm">
              <h3 className="font-extrabold text-slate-900 text-base mb-3">Log Handover Note</h3>
              <form onSubmit={handleAddHandover} className="space-y-3">
                <div className="flex flex-col sm:flex-row gap-3">
                  <select
                    value={newHandoverType}
                    onChange={(e) => setNewHandoverType(e.target.value)}
                    className="px-3 py-2 bg-slate-50 border border-slate-200 rounded-xl text-xs font-bold focus:outline-none focus:ring-2 focus:ring-emerald-500"
                  >
                    <option value="complaints">Guest Complaints</option>
                    <option value="mistakes">Kitchen Mistakes</option>
                    <option value="requests">Special Requests</option>
                  </select>
                  <input
                    type="text"
                    value={newHandoverText}
                    onChange={(e) => setNewHandoverText(e.target.value)}
                    placeholder="Enter details for handover..."
                    className="flex-1 px-4 py-2 bg-slate-50 border border-slate-200 rounded-xl text-xs font-medium focus:outline-none focus:ring-2 focus:ring-emerald-500"
                  />
                  <button
                    type="submit"
                    className="px-5 py-2 bg-emerald-600 hover:bg-emerald-700 text-white font-bold rounded-xl text-xs flex items-center justify-center gap-2 transition-colors"
                  >
                    <Plus className="w-4 h-4" /> Add Note
                  </button>
                </div>
              </form>
            </div>

            {/* 3 Organized Panels */}
            <div className="grid grid-cols-1 md:grid-cols-3 gap-6">
              
              {/* Guest Complaints */}
              <div className="bg-white p-5 rounded-3xl border border-red-200 shadow-sm">
                <div className="flex items-center gap-2 pb-3 border-b border-red-100 mb-4">
                  <div className="p-2 bg-red-100 rounded-xl text-red-700">
                    <AlertCircle className="w-5 h-5" />
                  </div>
                  <h3 className="font-bold text-slate-900 text-base">Guest Complaints</h3>
                </div>

                <div className="space-y-3">
                  {handovers.complaints.map((item) => (
                    <div key={item.id} className="p-3 bg-red-50/50 rounded-2xl border border-red-100 space-y-1">
                      <p className="text-xs font-medium text-slate-800 leading-relaxed">{item.text}</p>
                      <div className="flex items-center justify-between text-[10px] text-slate-400 pt-1 font-semibold">
                        <span>{item.reporter}</span>
                        <span>{item.time}</span>
                      </div>
                    </div>
                  ))}
                </div>
              </div>

              {/* Kitchen Mistakes */}
              <div className="bg-white p-5 rounded-3xl border border-amber-200 shadow-sm">
                <div className="flex items-center gap-2 pb-3 border-b border-amber-100 mb-4">
                  <div className="p-2 bg-amber-100 rounded-xl text-amber-700">
                    <AlertTriangle className="w-5 h-5" />
                  </div>
                  <h3 className="font-bold text-slate-900 text-base">Kitchen Mistakes</h3>
                </div>

                <div className="space-y-3">
                  {handovers.mistakes.map((item) => (
                    <div key={item.id} className="p-3 bg-amber-50/50 rounded-2xl border border-amber-100 space-y-1">
                      <p className="text-xs font-medium text-slate-800 leading-relaxed">{item.text}</p>
                      <div className="flex items-center justify-between text-[10px] text-slate-400 pt-1 font-semibold">
                        <span>{item.reporter}</span>
                        <span>{item.time}</span>
                      </div>
                    </div>
                  ))}
                </div>
              </div>

              {/* Special Requests */}
              <div className="bg-white p-5 rounded-3xl border border-blue-200 shadow-sm">
                <div className="flex items-center gap-2 pb-3 border-b border-blue-100 mb-4">
                  <div className="p-2 bg-blue-100 rounded-xl text-blue-700">
                    <Sparkles className="w-5 h-5" />
                  </div>
                  <h3 className="font-bold text-slate-900 text-base">Special Requests</h3>
                </div>

                <div className="space-y-3">
                  {handovers.requests.map((item) => (
                    <div key={item.id} className="p-3 bg-blue-50/50 rounded-2xl border border-blue-100 space-y-1">
                      <p className="text-xs font-medium text-slate-800 leading-relaxed">{item.text}</p>
                      <div className="flex items-center justify-between text-[10px] text-slate-400 pt-1 font-semibold">
                        <span>{item.reporter}</span>
                        <span>{item.time}</span>
                      </div>
                    </div>
                  ))}
                </div>
              </div>

            </div>
          </div>
        )}

      </main>

      {}
      {/* Photo / Video Modal */}
      {modalMedia && (
        <div className="fixed inset-0 z-50 bg-slate-900/80 backdrop-blur-xs flex items-center justify-center p-4">
          <div className="bg-white rounded-3xl max-w-2xl w-full overflow-hidden shadow-2xl relative animate-in fade-in zoom-in-95">
            <div className="p-4 bg-slate-900 text-white flex items-center justify-between">
              <h3 className="font-bold text-sm">{modalMedia.title}</h3>
              <button
                onClick={() => setModalMedia(null)}
                className="p-1 rounded-full hover:bg-slate-800 text-slate-400 hover:text-white"
              >
                <X className="w-5 h-5" />
              </button>
            </div>
            <div className="p-4 bg-black flex items-center justify-center min-h-[300px]">
              {modalMedia.type === 'image' ? (
                <img src={modalMedia.url} alt={modalMedia.title} className="max-h-[70vh] object-contain rounded-xl" />
              ) : (
                <video controls autoPlay className="w-full max-h-[70vh] rounded-xl">
                  <source src={modalMedia.url} type="video/mp4" />
                  Your browser does not support video playback.
                </video>
              )}
            </div>
          </div>
        </div>
      )}

      {}
      {/* WHATSAPP-STYLE BOTTOM NAVIGATION BAR */}
      <nav className="fixed bottom-0 left-0 right-0 bg-white border-t border-slate-200 z-40 py-2 px-3 shadow-lg">
        <div className="max-w-xl mx-auto flex items-center justify-around">
          
          {/* 1. Schedule */}
          <button
            onClick={() => setActiveTab('schedule')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'schedule'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">📅</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'schedule'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              Schedule
            </span>
          </button>

          {/* 2. Daily Duties */}
          <button
            onClick={() => setActiveTab('duties')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'duties'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">📌</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'duties'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              Daily Duties
            </span>
          </button>

          {/* 3. SOPs */}
          <button
            onClick={() => setActiveTab('sops')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'sops'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">🥗</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'sops'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              SOPs
            </span>
          </button>

          {/* 4. Shift Feedback */}
          <button
            onClick={() => setActiveTab('feedback')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'feedback'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">📝</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'feedback'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              Feedback
            </span>
          </button>

          {/* 5. Inventory */}
          <button
            onClick={() => setActiveTab('inventory')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'inventory'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">📦</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'inventory'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              Inventory
            </span>
          </button>

          {/* 6. Handover */}
          <button
            onClick={() => setActiveTab('handover')}
            className="flex flex-col items-center gap-1 group focus:outline-none"
          >
            <div
              className={`px-4 py-1 rounded-full transition-all flex items-center justify-center ${
                activeTab === 'handover'
                  ? 'bg-emerald-100 text-emerald-900'
                  : 'text-slate-600 hover:text-slate-900'
              }`}
            >
              <span className="text-xl">📋</span>
            </div>
            <span
              className={`text-[11px] transition-colors ${
                activeTab === 'handover'
                  ? 'font-black text-slate-900'
                  : 'font-semibold text-slate-500'
              }`}
            >
              Handover
            </span>
          </button>

        </div>
      </nav>
    </div>
  );
}
